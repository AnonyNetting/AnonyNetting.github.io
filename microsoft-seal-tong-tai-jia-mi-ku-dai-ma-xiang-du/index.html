    Microsoft SEAL同态加密库代码详读

    发布于 2025-04-30  57 次阅读

    Microsoft SEAL是由微软开发的开源同态加密算法库，目前支持BGV, BFV, CKKS以及其RNS下的同态操作，提供8个examples支持使用者直观学习使用方法，全部examples不使用RNS，但是库中包含RNS相关函数。这里预计先写2, 5, 6的代码理解，搭配同态加密(Homomorphic Encryption, HE)调查兼CKKS数篇文章带读可能方便理解。另外给SEAL evaluator.cpp中提供的没有出现在以上examples的算子写一份测试用例，因为发现SEAL存在少量参数选择很不主流，调试时可能出坑，例如编码时选取的基与HEAAN和后来的论文都不同，SEAL选择3而一般文献都选择5，这影响了替换操作和旋转操作的galois key在何值处生成能取得等价性。后续应该还会写一份自己的RNS-CKKS example。(使用matlab编写了一份RNS-CKKS，仓库在https://github.com/AnonyNetting/CKKS_matlab.git）

    2. encoders

    2_encoders.cpp其实包含面向BFV, BGV的batch_encoder和CKKS的ckks_encoder。这里只谈ckks_encoder。

    目标问题：

    # of slot: 4096，未编码明文x = [0.0, 1.1, 2.2, 3.3]，同态下求解x^2.

    代码

    void example_ckks_encoder()
{
    print_example_banner("Example: Encoders / CKKS Encoder");

    EncryptionParameters parms(scheme_type::ckks);//声明面向ckks的参数结构parms

    size_t poly_modulus_degree = 8192;//多项式order = 8192
    parms.set_poly_modulus_degree(poly_modulus_degree);
    parms.set_coeff_modulus(CoeffModulus::Create(poly_modulus_degree, { 40, 40, 40, 40, 40 }));//数字表示bit宽度，在这个宽度的附近用CoeffModulus::Create函数找到可使用的素数;从左到右依次对应于q, p0, p1, p2, P;如同开头所引链接文章内容，这里的位宽设置的合理逻辑是q=P>pi，但是这里没有设置q, P > pi的原因是任务x^2只会降低一次层级，而不会使密文模降低到仅剩下q导致模的数量级靠近缩放因子，所以不需要像通用做法一样安排模数

    //以上设置parms被进一步调整结构用于生成公私钥和重线性密钥
    SEALContext context(parms);
    print_parameters(context);
    cout << endl;

    /*
    Keys are created the same way as for the BFV scheme.
    */
    KeyGenerator keygen(context);
    auto secret_key = keygen.secret_key();
    PublicKey public_key;
    keygen.create_public_key(public_key);
    RelinKeys relin_keys;
    keygen.create_relin_keys(relin_keys);

    //加密器，计算器，解密器
    Encryptor encryptor(context, public_key);
    Evaluator evaluator(context);
    Decryptor decryptor(context, secret_key);

    //编码器
    CKKSEncoder encoder(context);

    //# of slots in ckks is N/2，在这里# of slots = 4192
    size_t slot_count = encoder.slot_count();
    cout << "Number of slots: " << slot_count << endl;

    //待编码的向量长度<N/2时，会自动为剩余位置补0，比如这里长度仅4，剩余4192-4都会补0
    vector<double> input{ 0.0, 1.1, 2.2, 3.3 };
    cout << "Input vector: " << endl;
    print_vector(input);

    //注意到scale = 2^30，与q并不处在同一数量级上，这是因为2_encoder.cpp本身就不严格符合ckks，其实更建议去看5_ckks_basics.cpp
    Plaintext plain;
    double scale = pow(2.0, 30);
    print_line(__LINE__);
    cout << "Encode input vector." << endl;
    encoder.encode(input, scale, plain);

    //立即执行一次解码，查看编码效果
    vector<double> output;
    cout << "    + Decode input vector ...... Correct." << endl;
    encoder.decode(plain, output);
    print_vector(output);

    //加密
    Ciphertext encrypted;
    print_line(__LINE__);
    cout << "Encrypt input vector, square, and relinearize." << endl;
    encryptor.encrypt(plain, encrypted);

    /*
    接下来就是在密文上平方，解密，解码；解码结果向量长度N/2，这是因为编码时补0到N/2的长度
    */
    evaluator.square_inplace(encrypted);
    evaluator.relinearize_inplace(encrypted, relin_keys);

    /*
    平方也就是自乘，这导致缩放因子变为2^60，正常来说需要一次重缩放；但这里不是正常的CKKS算法演示，于是这里的做法是在解密时去掉缩放因子的步骤将缩放因子用2^60而不是最初的2^30来处理
    */
    cout << "    + Scale in squared input: " << encrypted.scale() << " (" << log2(encrypted.scale()) << " bits)"
         << endl;

    print_line(__LINE__);
    cout << "Decrypt and decode." << endl;
    decryptor.decrypt(encrypted, plain);
    encoder.decode(plain, output);
    cout << "    + Result vector ...... Correct." << endl;
    print_vector(output);

    /*
    正常来说CKKS是允许在同态下降低缩放因子的。这将在3_levels.cpp和5_ckks_basics.cpp中阐述
    */
}

    5. CKKS Basics

    目标问题：

    # of slot: 4096，在[0, 1]区间上均匀取4096个点，对所有点同态下执行 PIx^3 + 0.4x + 1.

    代码

    void example_ckks_basics()
{
    print_example_banner("template: ckks basic function");

    EncryptionParameters parms(scheme_type::ckks);//表示CKKS参数的集合

    size_t poly_modulus_degree = 8192;//指定多项式环order = 8192
    parms.set_poly_modulus_degree(poly_modulus_degree);
    parms.set_coeff_modulus(CoeffModulus::Create(poly_modulus_degree, { 60, 40, 40, 60 }));//数字表示bit宽度，在这个宽度的附近用CoeffModulus::Create函数找到可使用的素数;从左向右依次对应q0, p1, p2, P;如同开头所引链接文章内容，这里的位宽设置逻辑是q=P>pi。
    double scale = pow(2.0, 40);//缩放因子设置靠近pi，与引文描述相同

//使用这些参数设计开始生成相关密钥，包括加解密钥和其他辅助密钥
    SEALContext context(parms);
    print_parameters(context);
    cout << endl;

    KeyGenerator keygen(context);
    auto secret_key = keygen.secret_key();
    PublicKey public_key;
    keygen.create_public_key(public_key);
    RelinKeys relin_keys;
    keygen.create_relin_keys(relin_keys);
    GaloisKeys gal_keys;
    keygen.create_galois_keys(gal_keys);
//定义加密器，计算器，解密器，编码器
    Encryptor encryptor(context, public_key);
    Evaluator evaluator(context);
    Decryptor decryptor(context, secret_key);
    CKKSEncoder encoder(context);
    size_t slot_count = encoder.slot_count();
    cout << "Number of slot" << slot_count << endl;//这里就是8192/2

    vector<double> input;
    input.reserve(slot_count);
    double curr_point = 0;
    double step_size = 1.0 / (static_cast<double>(slot_count) - 1);
    for (size_t i = 0; i < slot_count; i++)
    {
        input.push_back(curr_point);
        curr_point += step_size;
    }
    cout << "input vector" << endl;
    print_vector(input, 3, 7);

    cout << "Evaluating polynomial PI*x^3 + 0.4x + 1 ..." << endl;

    /*
    PI 0.4 1作为后续要参与乘法的明文也要执行encoder，但不必encrypt；以明文多项式方式参与的话实际上就是将自身重复N/2次填入所有slot；注意这里scale也被传入，即明文数值也应当在编码后被scale放大
    */
    Plaintext plain_coeff3, plain_coeff1, plain_coeff0;
    encoder.encode(3.14159265, scale, plain_coeff3);
    encoder.encode(0.4, scale, plain_coeff1);
    encoder.encode(1.0, scale, plain_coeff0);

    Plaintext x_plain;
    print_line(__LINE__);
    cout << "Encode input vectors" << endl;
    encoder.encode(input, scale, x_plain);//[0,1]上的N/2个点被编码
    Ciphertext x1_encrypted;
    encryptor.encrypt(x_plain, x1_encrypted);//加密器生成对应上面编码的密文

    /*
    要计算x^3，我们首先计算x^2并重线性化。计算得到的x^2对应的缩放因子已增长到2^80。
    */
    Ciphertext x3_encrypted;
    print_line(__LINE__);
    cout << "Compute x^2 and relinearize" << endl;
    evaluator.square(x1_encrypted, x3_encrypted);//使用x计算x^2，存进x^3的密文结构里面去
    evaluator.relinearize_inplace(x3_encrypted, relin_keys);//对x^3这个密文结构原址执行重线性化
    cout << "    + Scale of x^2 before rescale: " << log2(x3_encrypted.scale()) << "bits" << endl;

    /*
    现在进行重缩放；缩放因子会按被移除的素数（40位素数）等比例缩小。
    因此，新的缩放因子应接近2^40。但要注意，缩放因子从这个步骤后并不等于2^40，
    这是因为40位素数只是接近2^40。
    */
    print_line(__LINE__);
    cout << "Rescale x^2。" << endl;
    evaluator.rescale_to_next_inplace(x3_encrypted);//对x^3这个密文结构原址执行重缩放
    cout << "    + Scale of x^2 after rescale: " << log2(x3_encrypted.scale()) << "bits" << endl;

    /*
    现在x3_encrypted和x1_encrypted处于不同层级，这阻碍了我们通过相乘计算x^3。
    我们可以简单地将x1_encrypted切换到模切换链中的下一个参数。但由于我们仍需要将x^3项与PI（plain_coeff3）相乘，
    我们改为先计算PI*x，然后与x^2相乘得到PI*x^3。为此，我们计算PI*x并将其缩放因子从2^80重新调整到接近2^40。可见，明文×密文也是要借由重缩放来恢复的，即：明文×密文也会消耗深度
    */
    print_line(__LINE__);
    cout << "Compute and rescale PI*x." << endl;
    Ciphertext x1_encrypted_coeff3;
    evaluator.multiply_plain(x1_encrypted, plain_coeff3, x1_encrypted_coeff3);//multiply_plain专用于执行密文×明文
    cout << "    + Scale of PI*x before rescale: " << log2(x1_encrypted_coeff3.scale()) << "bits" << endl;
    evaluator.rescale_to_next_inplace(x1_encrypted_coeff3);
    cout << "    + Scale of PI*x after rescale: " << log2(x1_encrypted_coeff3.scale()) << "bits" << endl;

    /*
    由于x3_encrypted和x1_encrypted_coeff3具有完全相同的缩放因子并使用相同的加密参数，
    我们可以将它们相乘。我们将结果写入x3_encrypted，重线性化并重缩放。
    注意实际缩放因子在上次的基础上再次偏移（未必误差变大，只是再次发生变化）。
    此外，由于我们设计pi仅包含2级，在完成PI·x^3后已经消耗尽了深度
    */
    print_line(__LINE__);
    cout << "Compute, relinearize, and rescale (PI*x)*x^2。" << endl;
    evaluator.multiply_inplace(x3_encrypted, x1_encrypted_coeff3);
    evaluator.relinearize_inplace(x3_encrypted, relin_keys);
    cout << "    + Scale of PI*x^3 before rescale: " << log2(x3_encrypted.scale()) << "bits" << endl;
    evaluator.rescale_to_next_inplace(x3_encrypted);
    cout << "    + Scale of PI*x^3 after rescale: " << log2(x3_encrypted.scale()) << "bits" << endl;

    /*
    接下来我们计算一次项。这只需要一次与plain_coeff1的multiply_plain操作。
    我们用结果覆盖x1_encrypted。
    */
    print_line(__LINE__);
    cout << "Compute and rescale 0.4*x。" << endl;
    evaluator.multiply_plain_inplace(x1_encrypted, plain_coeff1);
    cout << "    + Scale of 0.4*x before rescale: " << log2(x1_encrypted.scale()) << "bits" << endl;
    evaluator.rescale_to_next_inplace(x1_encrypted);
    cout << "    + Scale of 0.4*x after rescale: " << log2(x1_encrypted.scale()) << " bits" << endl;

    /*
    现在要执行三项的合并。问题在于，现在这三项分别处在不同的深度上，对应了不同的模数。Evaluator会审查参与计算的对象的parms_id，如果不一致就会报错。这里先显示一下待合并的三项的所处在的深度
    */
    cout << endl;
    print_line(__LINE__);
    cout << "Parameters used by all three terms are different." << endl;
    cout << "    + Modulus chain index for x3_encrypted: "
         << context.get_context_data(x3_encrypted.parms_id())->chain_index() << endl;
    cout << "    + Modulus chain index for x1_encrypted: "
         << context.get_context_data(x1_encrypted.parms_id())->chain_index() << endl;
    cout << "    + Modulus chain index for plain_coeff0: "
         << context.get_context_data(plain_coeff0.parms_id())->chain_index() << endl;
    cout << endl;

    /*
    整理一下目前的深度是怎么来的

        - Product x^2 has scale 2^80 and is at level 2;
        - Product PI*x has scale 2^80 and is at level 2;
        - We rescaled both down to scale 2^80/P_2 and level 1;
        - Product PI*x^3 has scale (2^80/P_2)^2;
        - We rescaled it down to scale (2^80/P_2)^2/P_1 and level 0;
        - Product 0.4*x has scale 2^80;
        - We rescaled it down to scale 2^80/P_2 and level 1;
        - The contant term 1 has scale 2^40 and is at level 2.

    虽然它们的缩放因子都大约是 2^40, 但精确来说是不同的，因此也不能加在一起。这里显示一下各自实际使用的缩放因子
    */
    print_line(__LINE__);
    cout << "The exact scales of all three terms are different:" << endl;
    ios old_fmt(nullptr);
    old_fmt.copyfmt(cout);
    cout << fixed << setprecision(10);
    cout << "    + Exact scale in PI*x^3: " << x3_encrypted.scale() << endl;
    cout << "    + Exact scale in  0.4*x: " << x1_encrypted.scale() << endl;
    cout << "    + Exact scale in      1: " << plain_coeff0.scale() << endl;
    cout << endl;
    cout.copyfmt(old_fmt);

    /*
    有不同的方法解决这个问题，例如①直接将PI*x^3和0.4*x的缩放因子标签改为2^40的初始精准值；这不会引起明显的偏差；

    还有方法例如②将1编码时使用2^80/p2，然后在密文下再与0.4*x乘一次，使其也落到深度0。这种方法需要额外的来自于加密方的操作，其实不太有用。

    下面用①来计算
    修改PI*x^3 and 0.4*x 的缩放因子到 2^40.
    */
    print_line(__LINE__);
    cout << "Normalize scales to 2^40." << endl;
    x3_encrypted.scale() = pow(2.0, 40);
    x1_encrypted.scale() = pow(2.0, 40);

    /*
    Evaluator会审查的parms_id不只是缩放因子，还包括其他参数。考虑到PI*x^3的深度最低，直接通过mod_switch将0.4*x和1的模数降下来是最简单对齐其他参数的方法
    */
    print_line(__LINE__);
    cout << "Normalize encryption parameters to the lowest level." << endl;
    parms_id_type last_parms_id = x3_encrypted.parms_id();
    evaluator.mod_switch_to_inplace(x1_encrypted, last_parms_id);
    evaluator.mod_switch_to_inplace(plain_coeff0, last_parms_id);

    /*
    所有参数和缩放因子已经对齐
    */
    print_line(__LINE__);
    cout << "Compute PI*x^3 + 0.4*x + 1." << endl;
    Ciphertext encrypted_result;
    evaluator.add(x3_encrypted, x1_encrypted, encrypted_result);//密文加
    evaluator.add_plain_inplace(encrypted_result, plain_coeff0);//密文＋明文

    /*
    首先确认理想输出
    */
    Plaintext plain_result;
    print_line(__LINE__);
    cout << "Decrypt and decode PI*x^3 + 0.4x + 1." << endl;
    cout << "    + Expected result:" << endl;
    vector<double> true_result;
    for (size_t i = 0; i < input.size(); i++)
    {
        double x = input[i];
        true_result.push_back((3.14159265 * x * x + 0.4) * x + 1);
    }
    print_vector(true_result, 3, 7);

    /*
    解密，解码，得到同态下的输出
    */
    decryptor.decrypt(encrypted_result, plain_result);
    vector<double> result;
    encoder.decode(plain_result, result);
    cout << "    + Computed result ...... Correct." << endl;
    print_vector(result, 3, 7);

    /*
    这里没有使用复数作为输入用例，但是CKKS支持复数输入
    */
}

    6. Rotation

    目标问题：

    # of slot: 4096，在[0, 1]区间上均匀取4096个点，对[0 1/4095 ... 4095/4095]同态下执行向左旋转（循环移位）2个slot.

    代码

    void example_rotation_ckks()
{
    print_example_banner("Example: Rotation / Rotation in CKKS");

    EncryptionParameters parms(scheme_type::ckks);

    size_t poly_modulus_degree = 8192;
    parms.set_poly_modulus_degree(poly_modulus_degree);
    parms.set_coeff_modulus(CoeffModulus::Create(poly_modulus_degree, { 40, 40, 40, 40, 40 }));//这里的模数选取又没有遵守q = P > pi,因为确定不会掉落深度到0层

    SEALContext context(parms);
    print_parameters(context);
    cout << endl;

    KeyGenerator keygen(context);
    SecretKey secret_key = keygen.secret_key();
    PublicKey public_key;
    keygen.create_public_key(public_key);
    RelinKeys relin_keys;
    keygen.create_relin_keys(relin_keys);
    GaloisKeys galois_keys;
    keygen.create_galois_keys(galois_keys);
    Encryptor encryptor(context, public_key);
    Evaluator evaluator(context);
    Decryptor decryptor(context, secret_key);

    CKKSEncoder ckks_encoder(context);

    size_t slot_count = ckks_encoder.slot_count();
    cout << "Number of slots: " << slot_count << endl;
    vector<double> input;
    input.reserve(slot_count);
    double curr_point = 0;
    double step_size = 1.0 / (static_cast<double>(slot_count) - 1);
    for (size_t i = 0; i < slot_count; i++, curr_point += step_size)
    {
        input.push_back(curr_point);
    }
    cout << "Input vector:" << endl;
    print_vector(input, 3, 7);

    auto scale = pow(2.0, 50);//缩放因子也没有靠近pi

    print_line(__LINE__);
    cout << "Encode and encrypt." << endl;
    Plaintext plain;
    ckks_encoder.encode(input, scale, plain);
    Ciphertext encrypted;
    encryptor.encrypt(plain, encrypted);

    Ciphertext rotated;
    print_line(__LINE__);
    cout << "Rotate 2 steps left." << endl;
    evaluator.rotate_vector(encrypted, 2, galois_keys, rotated);//使用旋转密钥完成旋转操作
    cout << "    + Decrypt and decode ...... Correct." << endl;
    decryptor.decrypt(rotated, plain);
    vector<double> result;
    ckks_encoder.decode(plain, result);
    print_vector(result, 3, 7);

    /*
    CKKS还可以执行同态求槽上共轭的功能，这个在引文链接里也有提到，类似于旋转，操作流程都是：提前KSGen生成旋转/共轭所需的辅助密钥；在密文多项式上执行替换，旋转时将X替换为X3r，共轭时将X替换为X-1；替换后使用辅助密钥和KS函数，将密文返回到使用s作为私钥的状态
    */
}

    剩余算子用例

    根据evaluator.h，可以看到SEAL支持以下算子：

    negate的原理应该算是比较简单，c1 = (-as + m + e, a)，密文直接取反得(as - m - e, -a)，应用密钥得(-m - e)。

    void example_negate_ckks()
{
    print_example_banner("Example: Negate in CKKS");

    EncryptionParameters parms(scheme_type::ckks);

    size_t poly_modulus_degree = 8192;
    parms.set_poly_modulus_degree(poly_modulus_degree);
    parms.set_coeff_modulus(CoeffModulus::Create(poly_modulus_degree, { 40, 40, 40, 40, 40 }));

    SEALContext context(parms);
    print_parameters(context);
    cout << endl;

    KeyGenerator keygen(context);
    SecretKey secret_key = keygen.secret_key();
    PublicKey public_key;
    keygen.create_public_key(public_key);
    Encryptor encryptor(context, public_key);
    Evaluator evaluator(context);
    Decryptor decryptor(context, secret_key);

    CKKSEncoder ckks_encoder(context);

    size_t slot_count = ckks_encoder.slot_count();
    cout << "Number of slots: " << slot_count << endl;
    vector<double> input;
    input.reserve(slot_count);
    double curr_point = 0;
    double step_size = 1.0 / (static_cast<double>(slot_count) - 1);
    for (size_t i = 0; i < slot_count; i++, curr_point += step_size)
    {
        input.push_back(curr_point);
    }
    cout << "Input vector:" << endl;
    print_vector(input, 3, 7);

    auto scale = pow(2.0, 50);

    print_line(__LINE__);
    cout << "Encode and encrypt." << endl;
    Plaintext plain;
    ckks_encoder.encode(input, scale, plain);
    Ciphertext encrypted;
    encryptor.encrypt(plain, encrypted);

    Ciphertext negated;
    print_line(__LINE__);
    cout << "Negate input" << endl;
    evaluator.negate(encrypted, negated);
    cout << "    + Decrypt and decode ...... Correct." << endl;
    decryptor.decrypt(negated, plain);
    vector<double> result;
    ckks_encoder.decode(plain, result);
    print_vector(result, 3, 7);
}

    mod_switch的功能与rescale类似，但是mod_swicth不调整密文所对应的缩放因子，即.scale()。mod_swicth函数其实对应CKKS原始论文Homomorphic Encryption for Arithmetic of Approximate Numbers中图3的右边，单纯将多项式上所有在原Q上取模的系数重新在层级更低的Q'上取模。对比rescale则需要对所有系数乘以要去掉的那一层所对应剩余模的逆，如下式；mod_switch则仅对应红框标识的部分。

    void example_mod_switch_ckks()
{
    print_example_banner("Example: Negate in CKKS");

    EncryptionParameters parms(scheme_type::ckks);
    size_t N = 8192;
    parms.set_poly_modulus_degree(N);
    parms.set_coeff_modulus(CoeffModulus::Create(N, {40, 40, 40, 40}));
    SEALContext context(parms);

    KeyGenerator keygen(context);
    SecretKey secret_key = keygen.secret_key();
    PublicKey public_key;
    keygen.create_public_key(public_key);
    RelinKeys relin_keys;
    keygen.create_relin_keys(relin_keys);
    GaloisKeys galois_keys;
    keygen.create_galois_keys(galois_keys);
    Encryptor encryptor(context, public_key);
    Evaluator evaluator(context);
    Decryptor decryptor(context, secret_key);

    CKKSEncoder ckks_encoder(context);

    size_t slot_count = ckks_encoder.slot_count();
    cout << "Number of slots: " << slot_count << endl;
    vector<double> input;
    input.reserve(slot_count);
    double curr_point = 0;
    double step_size = 1.0 / (static_cast<double>(slot_count) - 1);
    for (size_t i = 0; i < slot_count; i++, curr_point += step_size)
    {
        input.push_back(curr_point);
    }
    cout << "Input vector:" << endl;
    print_vector(input, 3, 7);

    auto scale = pow(2.0, 50);

    print_line(__LINE__);
    cout << "Encode and encrypt." << endl;
    Plaintext plain;
    ckks_encoder.encode(input, scale, plain);
    Ciphertext encrypted;
    encryptor.encrypt(plain, encrypted);

    cout << "Before mod_switch: Level (chain index) = " 
        << context.get_context_data(encrypted.parms_id())->chain_index() << endl;
    evaluator.mod_switch_to_next_inplace(encrypted);
    cout << "After mod_switch: Level (chain index) = " 
        << context.get_context_data(encrypted.parms_id())->chain_index() << endl;

    print_line(__LINE__);
    cout << "    + Decrypt and decode ...... Correct." << endl;
    decryptor.decrypt(encrypted, plain);
    vector<double> result;
    ckks_encoder.decode(plain, result);
    print_vector(result, 3, 7);
}

    rotate_row和rotate_column仅支持BFV， BGV，因为这两种算法加密对象是矩阵， rotate_column仅支持CKKS因为CKKS加密填充slot的vector。

    exponentiate执行同态求幂，但是SEAL不支持为CKKS执行这个操作，不知道是出于什么考虑。

    transform_to_ntt执行plain polynomial从normal到NTT域上的变换，便于multiply_plain。留意multiply_plain其实已经内含了这个功能（下图evaluator.cpp可见已经考虑了multiply_plain时所有可能情况），明文可以直接以normal输入给multiply_plain，这个函数单列出来主要还是提供一个观察的效果。另外，transform_to_ntt是根据RNS分解在每个小因数上的环而进行的。

    apply_galois是非常重要的函数。以apply_galois_inplace(encrypted, i)为例，它将编码后多项式p(X)替换为p(X^i)。如前面分析过的，在SEAL中，rotate(k)等价于apply_galois(3^k mod N)，conjugate等价于apply_galois(2N-1)。注意无论是apply_galois，还是其特殊情况rotate, conjugate都需要galois_key。galois_key需要通过keygen.create_galois_keys(galois_elts, galois_keys)函数生成。在仅使用rotate, conjugate的情况下可以省略galois_elts参数设置，而当我们需要替换X为任意X^i时则需要为其设置参数。以下代码给出用例。

    void example_apply_galois_ckks()
{
    print_example_banner("Example: Rotation / Rotation in CKKS");

    EncryptionParameters parms(scheme_type::ckks);

    size_t poly_modulus_degree = 8192;
    parms.set_poly_modulus_degree(poly_modulus_degree);
    parms.set_coeff_modulus(CoeffModulus::Create(poly_modulus_degree, { 40, 40, 40, 40, 40 }));

    SEALContext context(parms);
    print_parameters(context);
    cout << endl;

    KeyGenerator keygen(context);
    SecretKey secret_key = keygen.secret_key();
    PublicKey public_key;
    keygen.create_public_key(public_key);
    RelinKeys relin_keys;
    keygen.create_relin_keys(relin_keys);
    GaloisKeys galois_keys;
    //keygen.create_galois_keys(galois_keys);
    std::vector<std::uint32_t> galois_elts = {3, 2*8192-1};
    keygen.create_galois_keys(galois_elts, galois_keys);
    Encryptor encryptor(context, public_key);
    Evaluator evaluator(context);
    Decryptor decryptor(context, secret_key);

    CKKSEncoder ckks_encoder(context);

    size_t slot_count = ckks_encoder.slot_count();
    cout << "Number of slots: " << slot_count << endl;
    vector<complex<double>> input(slot_count);
    double step = 1.0 / (slot_count - 1);  // 1/4095 when slot_count=4096

    for (size_t k = 0; k < slot_count; k++) {
        input[k] = complex<double>(k * step, k * step);  // 实部和虚部相同
    }
    cout << "Input vector:" << endl;
    print_vector(input, 5, 11);

    auto scale = pow(2.0, 50);

    print_line(__LINE__);
    cout << "Encode and encrypt." << endl;
    Plaintext plain1, plain2, plain3, plain4;
    /*
    plain1: apply_galois(encrypted, 3)
    plain2: rotate_vector(encrypted, 1)
    plain3: apply_galois(encrypted, 2*N-1)
    plain4: complex_conjugate(encrypted)
    
    we expect plain1 == plain2, plain3 == plain4
    */
    ckks_encoder.encode(input, scale, plain1);
    Ciphertext encrypted;
    encryptor.encrypt(plain1, encrypted);

    Ciphertext encrypted1, encrypted2, encrypted3, encrypted4;
    print_line(__LINE__);
    evaluator.apply_galois(encrypted, 3, galois_keys, encrypted1);
    evaluator.rotate_vector(encrypted, 1, galois_keys, encrypted2);
    evaluator.apply_galois(encrypted, 2*8192-1, galois_keys, encrypted3);
    evaluator.complex_conjugate(encrypted, galois_keys, encrypted4);
    vector<complex<double>> result1, result2, result3, result4;
    cout << "Substitute x -> x^3:" << endl;
    decryptor.decrypt(encrypted1, plain1);
    ckks_encoder.decode(plain1, result1);
    print_vector(result1, 5, 11);
    cout << "Rotate 1:" << endl;
    decryptor.decrypt(encrypted2, plain2);
    ckks_encoder.decode(plain2, result2);
    print_vector(result2, 5, 11);
    cout << "Substitute x -> x^(2*N-1):" << endl;
    decryptor.decrypt(encrypted3, plain3);
    ckks_encoder.decode(plain3, result3);
    print_vector(result3, 5, 11);
    cout << "Complex conjugate:" << endl;
    decryptor.decrypt(encrypted4, plain4);
    ckks_encoder.decode(plain4, result4);
    print_vector(result4, 5, 11);
}
